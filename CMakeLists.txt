cmake_minimum_required(VERSION 3.20)
project(EngineSense VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(ENGINESENSE_BUILD_TESTS "Build GoogleTest unit/integration/regression tests" ON)
option(ENGINESENSE_BUILD_APPS "Build the CLI and benchmark tools" ON)
option(ENGINESENSE_BUILD_DESKTOP "Build the Qt 6 desktop application (needs Qt6 Widgets)" OFF)
option(ENGINESENSE_WERROR "Treat warnings as errors" OFF)
option(ENGINESENSE_CLANG_TIDY "Run clang-tidy during the build" OFF)
set(ENGINESENSE_SANITIZER "" CACHE STRING "address | thread | (empty). 'address' also enables UBSan.")

if(ENGINESENSE_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
elseif(ENGINESENSE_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()

if(ENGINESENSE_CLANG_TIDY)
  find_program(CLANG_TIDY_EXE NAMES clang-tidy REQUIRED)
endif()

add_subdirectory(core)

if(ENGINESENSE_BUILD_APPS)
  add_subdirectory(apps/cli)
  add_subdirectory(apps/bench)
endif()
if(ENGINESENSE_BUILD_DESKTOP)
  add_subdirectory(apps/desktop)
endif()
if(ENGINESENSE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
